-- §A1.3 — Views (regular + materialized) touching workstreams or meetings
| schema | name                       | kind | definition                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                               |
| ------ | -------------------------- | ---- | -------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------------- |
| public | meeting_attendees_resolved | view |  SELECT ma.id,
    ma.meeting_id,
    ma.attendance_status,
    ma.is_required,
    ma.notified_at,
        CASE
            WHEN ma.user_id IS NOT NULL THEN 'user'::text
            WHEN ma.stakeholder_id IS NOT NULL THEN 'stakeholder'::text
            ELSE 'external_contact'::text
        END AS attendee_type,
    COALESCE(u.name, (s.first_name || ' '::text) || s.last_name, (ec.first_name || ' '::text) || ec.last_name) AS full_name,
    COALESCE(u.email, s.email, ec.email) AS email,
    COALESCE(u.title, s.title, ec.title) AS title,
    COALESCE(s.company_name, ec.company_name) AS company_name,
    ma.user_id,
    ma.stakeholder_id,
    ma.external_contact_id
   FROM meeting_attendees ma
     LEFT JOIN users u ON ma.user_id = u.id
     LEFT JOIN stakeholders s ON ma.stakeholder_id = s.id
     LEFT JOIN external_contacts ec ON ma.external_contact_id = ec.id; |
| public | v_meeting_agenda_items     | view |  SELECT id,
    meeting_id,
    title,
    description,
    sequence_order AS sort_order,
    notes_captured AS notes,
    include_in_minutes,

    is_flagged,

   FROM meeting_agenda_items a;                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                                    |
| public | v_meetings                 | view |  SELECT m.id,
    m.firm_id,
    m.title,
    m.scheduled_date AS meeting_date,
    m.scheduled_duration_minutes AS duration_minutes,
    m.location AS location_or_link,
    m.meeting_type,
    m.status,
    m.organizer_id,
    m.project_id,
    m.prospect_id,
    m.prior_meeting_id AS parent_meeting_id,
    m.purpose,
    m.outcome,
    m.work_stream,
    m.client_id,
    p.name AS project_name,
    p.client_id AS project_client_id
   FROM meetings m
     LEFT JOIN projects p ON p.id = m.project_id;                                                                                                                                                                                                                                                                                                                                                                                |
